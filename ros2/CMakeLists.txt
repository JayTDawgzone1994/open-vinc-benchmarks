cmake_minimum_required(VERSION 3.20)
project(open_vinc_benchmark_ros LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosidl_default_generators REQUIRED)

set(message_names Frame Frame1k Frame1m Frame8m Frame16m Frame32m Frame64m Frame128m)
set(message_files)
foreach(name IN LISTS message_names)
    list(APPEND message_files "msg/${name}.msg")
endforeach()
rosidl_generate_interfaces(${PROJECT_NAME} ${message_files})
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)

# Separate typed instantiations keep compiler memory bounded for large schemas.
set(workload_objects)
foreach(name IN LISTS message_names)
    string(TOLOWER "${name}" lower)
    add_library(workload_${lower} OBJECT src/typed_workload.cpp)
    target_include_directories(workload_${lower} PRIVATE include)
    target_compile_definitions(workload_${lower} PRIVATE
        BENCH_MESSAGE_HEADER="open_vinc_benchmark_ros/msg/${lower}.hpp"
        BENCH_MESSAGE_TYPE=open_vinc_benchmark_ros::msg::${name}
        BENCH_FUNCTION=run_${lower})
    ament_target_dependencies(workload_${lower} rclcpp)
    target_link_libraries(workload_${lower} "${typesupport}")
    target_compile_options(workload_${lower} PRIVATE -Wall -Wextra -Wpedantic)
    list(APPEND workload_objects $<TARGET_OBJECTS:workload_${lower}>)
endforeach()
add_executable(ros_benchmark src/main.cpp ${workload_objects})
target_include_directories(ros_benchmark PRIVATE include)
ament_target_dependencies(ros_benchmark rclcpp)
target_link_libraries(ros_benchmark "${typesupport}")
target_compile_options(ros_benchmark PRIVATE -Wall -Wextra -Wpedantic)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/ros-build.json" CONTENT
"{\n  \"configuration\": \"$<CONFIG>\",\n  \"compiler\": \"${CMAKE_CXX_COMPILER_ID}\",\n  \"compiler_version\": \"${CMAKE_CXX_COMPILER_VERSION}\",\n  \"cmake_version\": \"${CMAKE_VERSION}\",\n  \"system\": \"${CMAKE_SYSTEM_NAME}\",\n  \"processor\": \"${CMAKE_SYSTEM_PROCESSOR}\"\n}\n")
install(TARGETS ros_benchmark DESTINATION lib/${PROJECT_NAME})
install(FILES ../LICENSE ../NOTICE DESTINATION share/${PROJECT_NAME})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ros-build.json" DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
